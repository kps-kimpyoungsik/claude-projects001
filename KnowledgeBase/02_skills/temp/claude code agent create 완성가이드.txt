{"task_id":"...","status":"done|failed|blocked","artifacts":["경로"],"message":"한줄요약"}

pythonimport json
import re
from pathlib import Path
from datetime import datetime
from dataclasses import dataclass, field

MEMORY_DIR   = Path("memory")
KNOWHOW_FILE = MEMORY_DIR / "knowhow.json"
PATTERN_FILE = MEMORY_DIR / "error_patterns.json"

@dataclass
class Issue:
    task_id:     str
    agent:       str
    stage:       str
    error_type:  str
    severity:    str
    raw_error:   str
    auto_fix:    str
    knowhow_id:  str | None
    detected_at: str = field(default_factory=lambda: datetime.now().isoformat())


class IssueAnalyzer:
    def __init__(self):
        self.patterns = json.loads(PATTERN_FILE.read_text())["patterns"]
        self.knowhow  = json.loads(KNOWHOW_FILE.read_text())["entries"]

    def analyze(self, task_id, agent, stage, error_text) -> Issue | None:
        for p in self.patterns:
            if re.search(p["regex"], error_text, re.IGNORECASE):
                return Issue(
                    task_id=task_id, agent=agent, stage=stage,
                    error_type=p["category"], severity=p["severity"],
                    raw_error=error_text[:300], auto_fix=p["auto_fix"],
                    knowhow_id=p.get("knowhow_ref")
                )
        return None

    def get_knowhow(self, issue: Issue) -> dict | None:
        if issue.knowhow_id:
            kh = next((k for k in self.knowhow if k["id"] == issue.knowhow_id), None)
            if kh:
                return kh
        return next(
            (k for k in self.knowhow if k["error_type"] == issue.error_type), None
        )

    def get_knowhow_by_stage(self, stage: str) -> list[dict]:
        return [k for k in self.knowhow if k["stage"] == stage
                and k.get("success_rate") and k["success_rate"] > 0.7]

    def record_result(self, issue: Issue, resolved: bool, solution: str):
        data = json.loads(KNOWHOW_FILE.read_text())
        for entry in data["entries"]:
            if entry["error_type"] == issue.error_type:
                cnt = entry["applied_count"] + 1
                prev = entry["success_rate"] or 0
                entry["applied_count"] = cnt
                entry["success_rate"]  = round((prev*(cnt-1) + int(resolved)) / cnt, 3)
                KNOWHOW_FILE.write_text(json.dumps(data, indent=2, ensure_ascii=False))
                return
        # 신규 추가
        new_id = f"kh-{len(data['entries'])+1:03d}"
        data["entries"].append({
            "id": new_id, "stage": issue.stage,
            "error_type": issue.error_type, "trigger": "자동 감지",
            "symptom": issue.raw_error[:200], "solution": solution,
            "prevention_rule": f"{issue.auto_fix} 자동 적용",
            "applied_count": 1, "success_rate": 1.0 if resolved else 0.0,
            "created_at": datetime.now().isoformat(),
            "tags": [issue.stage, issue.error_type]
        })
        KNOWHOW_FILE.write_text(json.dumps(data, indent=2, ensure_ascii=False))
        print(f"  📚 새 노하우 저장: {new_id} ({issue.error_type})")
optimizer/applier.py
pythonimport json
import asyncio
from pathlib import Path
from datetime import datetime
from optimizer.analyzer import IssueAnalyzer, Issue

AGENT_CLAUDE_MD = {
    "dev":    Path("agents/dev/CLAUDE.md"),
    "test":   Path("agents/test/CLAUDE.md"),
    "review": Path("agents/review/CLAUDE.md"),
    "docs":   Path("agents/docs/CLAUDE.md"),
}
OPT_LOG = Path("memory/optimizations.json")


class AutoOptimizer:
    def __init__(self):
        self.analyzer = IssueAnalyzer()

    async def handle(self, task: dict, error_text: str) -> dict:
        issue = self.analyzer.analyze(task["id"], task["role"], task["role"], error_text)
        if not issue:
            return {"action": "none", "success": False}

        print(f"\n  🔍 이슈 감지: [{issue.severity}] {issue.error_type}")
        knowhow = self.analyzer.get_knowhow(issue)
        if knowhow:
            print(f"  📖 노하우: {knowhow['id']} - {knowhow['solution'][:60]}")

        fix_map = {
            "reduce_scope":        self._fix_scope,
            "verify_artifacts":    self._fix_artifacts,
            "exponential_backoff": self._fix_backoff,
            "check_settings_json": self._fix_permissions,
            "split_task":          self._fix_split,
        }
        handler = fix_map.get(issue.auto_fix, self._fix_generic)
        result  = await handler(issue, task, knowhow)
        self._log(issue, result, knowhow)
        self.analyzer.record_result(issue, result.get("success", False), result.get("solution",""))
        return result

    async def _fix_scope(self, issue, task, knowhow):
        md = AGENT_CLAUDE_MD.get(issue.agent)
        if not md or not md.exists():
            return {"action": "reduce_scope", "success": False}
        content = md.read_text()
        guard = "\n## AUTO TOKEN GUARD\n- NEVER scan: node_modules,.git,dist,build,coverage,*.lock\n- MAX reads: 20\n"
        if "AUTO TOKEN GUARD" not in content:
            md.write_text(content + guard)
            print(f"  ✅ 스코프 축소 적용: {issue.agent}")
        return {"action": "reduce_scope", "success": True, "solution": "스코프 제한 추가"}

    async def _fix_artifacts(self, issue, task, knowhow):
        dep_id = task.get("depends_on")
        if not dep_id:
            return {"action": "verify_artifacts", "success": False}
        sf = Path("shared/status.json")
        status = json.loads(sf.read_text())
        artifacts = status.get(dep_id, {}).get("artifacts", [])
        missing = [a for a in artifacts if not Path(a).exists()]
        if missing:
            print(f"  ⚠️  누락: {missing}")
            status[dep_id]["needs_retry"] = True
            sf.write_text(json.dumps(status, indent=2))
            return {"action": "verify_artifacts", "success": False, "retry_dep": dep_id}
        return {"action": "verify_artifacts", "success": True, "solution": "경로 확인 완료"}

    async def _fix_backoff(self, issue, task, knowhow):
        wait = 30
        print(f"  ⏳ Rate limit - {wait}s 대기...")
        await asyncio.sleep(wait)
        return {"action": "exponential_backoff", "success": True, "solution": f"{wait}s 대기"}

    async def _fix_permissions(self, issue, task, knowhow):
        sf = Path(".claude/settings.json")
        s  = json.loads(sf.read_text()) if sf.exists() else {}
        s.setdefault("permissions", {"allow": []})
        for p in ["Bash(*)", "Read(*)", "Write(*)", "Edit(*)"]:
            if p not in s["permissions"]["allow"]:
                s["permissions"]["allow"].append(p)
        sf.write_text(json.dumps(s, indent=2))
        return {"action": "check_settings_json", "success": True, "solution": "권한 추가"}

    async def _fix_split(self, issue, task, knowhow):
        prompt = f"다음 태스크를 2개 서브태스크로 분할. JSON 배열만 응답:\n{task['instruction']}"
        proc = await asyncio.create_subprocess_exec(
            "claude", "-p", prompt,
            stdout=asyncio.subprocess.PIPE
        )
        stdout, _ = await proc.communicate()
        try:
            subtasks = json.loads(stdout.decode().strip())
            tf = Path("shared/tasks.json")
            all_tasks = json.loads(tf.read_text())
            for i, st in enumerate(subtasks):
                st.update({"id": f"{task['id']}-sub{i+1}", "depends_on": None, "priority": task.get("priority",1)})
                all_tasks.append(st)
            tf.write_text(json.dumps(all_tasks, indent=2, ensure_ascii=False))
            return {"action": "split_task", "success": True}
        except Exception:
            return {"action": "split_task", "success": False}

    async def _fix_generic(self, issue, task, knowhow):
        return {"action": "generic", "success": False,
                "solution": knowhow["solution"] if knowhow else "수동 확인 필요"}

    def _log(self, issue, result, knowhow):
        data = json.loads(OPT_LOG.read_text()) if OPT_LOG.exists() else {"logs": []}
        data["logs"].append({
            "timestamp":  datetime.now().isoformat(),
            "task_id":    issue.task_id,
            "error_type": issue.error_type,
            "auto_fix":   issue.auto_fix,
            "knowhow_id": knowhow["id"] if knowhow else None,
            "success":    result.get("success", False)
        })
        OPT_LOG.write_text(json.dumps(data, indent=2, ensure_ascii=False))

pythonimport json
import asyncio
from pathlib import Path
from datetime import datetime
from optimizer.applier import AutoOptimizer

SHARED    = Path("shared")
TASKS_FILE  = SHARED / "tasks.json"
STATUS_FILE = SHARED / "status.json"

AGENT_CONFIG = {
    "dev":    {"worktree": "worktrees/agent-dev"},
    "test":   {"worktree": "worktrees/agent-test"},
    "review": {"worktree": "worktrees/agent-review"},
    "docs":   {"worktree": "worktrees/agent-docs"},
}

optimizer = AutoOptimizer()


def update_status(task_id: str, status: str, agent: str, extras: dict = {}):
    sf   = STATUS_FILE
    data = json.loads(sf.read_text()) if sf.exists() else {}
    data[task_id] = {
        "status": status, "agent": agent,
        "updated_at": datetime.now().isoformat(), **extras
    }
    sf.write_text(json.dumps(data, indent=2))


async def run_agent(task: dict, retry: int = 0) -> dict:
    MAX_RETRY = 3
    role = task["role"]
    cfg  = AGENT_CONFIG[role]

    # 사전 예방 노하우 주입
    prevention = ""
    hints = optimizer.analyzer.get_knowhow_by_stage(role)
    if hints:
        prevention = "\n# 주의사항 (과거 이슈 기반)\n" + "\n".join(f"- {h['prevention_rule']}" for h in hints[:2])

    prompt = (
        f"ROLE_FILE: agents/{role}/CLAUDE.md\n"
        f"CONTEXT: shared/context.json\n"
        f"TASK: {json.dumps(task, ensure_ascii=False)}\n"
        f"{prevention}\n"
        f"OUTPUT FORMAT JSON만 응답. 설명 생략."
    )

    update_status(task["id"], "running", role)

    proc = await asyncio.create_subprocess_exec(
        "claude", "-p", prompt,
        "--dangerously-skip-permissions",
        cwd=cfg["worktree"],
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.PIPE
    )
    stdout, stderr = await proc.communicate()
    output     = stdout.decode().strip()
    error_text = stderr.decode().strip()

    # 실패 감지 → 자동 최적화 → 재시도
    is_failed = proc.returncode != 0 or '"status":"failed"' in output
    if is_failed and retry < MAX_RETRY:
        fix = await optimizer.handle(task, error_text or output)
        if fix.get("success") or fix["action"] == "exponential_backoff":
            print(f"  🔁 재시도 {retry+1}/{MAX_RETRY}: {task['id']}")
            update_status(task["id"], "retrying", role)
            return await run_agent(task, retry + 1)

    try:
        result = json.loads(output)
    except json.JSONDecodeError:
        result = {"task_id": task["id"], "status": "failed", "message": output[:200]}

    update_status(task["id"], result.get("status","unknown"), role,
                  {"artifacts": result.get("artifacts", [])})
    return result


async def dispatch(tasks: list[dict]) -> list[dict]:
    independent = [t for t in tasks if not t.get("depends_on")]
    dependent   = [t for t in tasks if t.get("depends_on")]

    print(f"\n[Orchestrator] 병렬 실행: {[t['id'] for t in independent]}")
    results = list(await asyncio.gather(*[run_agent(t) for t in independent]))

    for task in sorted(dependent, key=lambda t: t.get("priority", 9)):
        dep_id = task["depends_on"]
        dep    = next((r for r in results if r.get("task_id") == dep_id), None)
        sf     = json.loads(STATUS_FILE.read_text())
        dep_st = sf.get(dep_id, {}).get("status")

        if dep_st == "done":
            print(f"[Orchestrator] 순차 실행: {task['id']} (의존 {dep_id} 완료)")
            result = await run_agent(task)
            results.append(result)
        else:
            print(f"[Orchestrator] 스킵: {task['id']} (의존 {dep_id} → {dep_st})")
            update_status(task["id"], "skipped", task["role"])

    return results

        except Exception:
            out[k] = {} if k != "tasks" else []
    return out


def render():
    d       = load()
    tasks   = d["tasks"] if isinstance(d["tasks"], list) else []
    status  = d["status"]
    knowhow = d["knowhow"].get("entries", [])
    opts    = d["opts"].get("logs", [])
    W = 68

    os.system("clear")
    print("═" * W)
    print(f"  🤖 Agent Team  [{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}]")
    print("═" * W)

    total    = len(tasks)
    done_n   = sum(1 for t in tasks if status.get(t["id"],{}).get("status")=="done")
    fail_n   = sum(1 for t in tasks if status.get(t["id"],{}).get("status")=="failed")
    run_n    = sum(1 for t in tasks if status.get(t["id"],{}).get("status")=="running")
    retry_n  = sum(1 for t in tasks if status.get(t["id"],{}).get("status")=="retrying")
    pct      = int(done_n/total*36) if total else 0
    bar      = "█"*pct + "░"*(36-pct)

    print(f"\n  [{bar}] {done_n}/{total}  🔄{run_n} 🔁{retry_n} ✅{done_n} ❌{fail_n}\n")

    # 태스크 테이블
    print(f"  {'TASK':<12} {'ROLE':<8} {'STATUS':<10} {'UPDATED':<17} DEPS")
    print(f"  {'─'*60}")
    for t in sorted(tasks, key=lambda x: x.get("priority",9)):
        s   = status.get(t["id"], {})
        st  = s.get("status","pending")
        ico = ST_ICON.get(st, "?")
        ts  = s.get("updated_at","-")[:16]
        dep = f"⬅ {t['depends_on']}" if t.get("depends_on") else ""
        print(f"  {t['id']:<12} {t['role']:<8} {ico} {st:<8} {ts}  {dep}")

    # 최적화 이력
    if opts:
        recent = opts[-4:]
        print(f"\n  {'─'*60}")
        print(f"  ⚡ 최적화 이력 (최근 {len(recent)}건)")
        for o in reversed(recent):
            ok = "✅" if o.get("success") else "❌"
            kh = f"[{o['knowhow_id']}]" if o.get("knowhow_id") else "[신규]"
            ts = o.get("timestamp","")[:16]
            print(f"  {ok} {ts}  {o['error_type']:<22}→ {o['auto_fix']:<18} {kh}")

    # 노하우 현황
    if knowhow:
        ranked = sorted(
            [k for k in knowhow if k.get("applied_count",0) > 0],
            key=lambda k: k["applied_count"], reverse=True
        )[:4]
        if ranked:
            print(f"\n  {'─'*60}")
            print(f"  📚 노하우 DB ({len(knowhow)}건)")
            for k in ranked:
                rate  = k.get("success_rate") or 0
                bar_w = int(rate*10)
                bar_s = "▓"*bar_w + "░"*(10-bar_w)
                print(f"  [{k['id']}] {k['error_type']:<22} 적용:{k['applied_count']:>3}회  [{bar_s}]{rate:.0%}")

    # 에이전트 성능
    stats = defaultdict(lambda: {"done":0,"failed":0,"total":0})
    for t in tasks:
        s = status.get(t["id"],{})
        r = t["role"]
        stats[r]["total"] += 1
        if s.get("status") == "done":   stats[r]["done"] += 1
        elif s.get("status") == "failed": stats[r]["failed"] += 1

    if any(v["total"] for v in stats.values()):
        print(f"\n  {'─'*60}")
        print(f"  📊 에이전트 성능")
        for role, v in stats.items():
            rate  = v["done"]/v["total"] if v["total"] else 0
            bar_w = int(rate*12)
            bar_s = "▓"*bar_w + "░"*(12-bar_w)
            print(f"  {role:<8} [{bar_s}] {rate:.0%}  ✅{v['done']} ❌{v['failed']}")

    failed_ids = [t["id"] for t in tasks if status.get(t["id"],{}).get("status")=="failed"]
    if failed_ids:
        print(f"\n  ⚠️  수동 확인 필요: {', '.join(failed_ids)}")

    print("\n" + "═" * W)
    return done_n < total or run_n > 0 or retry_n > 0


def start(interval: int = 2):
    while render():
        time.sleep(interval)
    render()
    print("\n  🎉 전체 완료!")


if __name__ == "__main__":
    start()

pythonimport json
import asyncio
from pathlib import Path
from datetime import datetime

CP_FILE  = Path("shared/checkpoint.json")
PRG_FILE = Path("shared/progress.json")
INTERVAL = 10  # N step마다 검증


class CheckpointManager:
    def __init__(self, interval: int = INTERVAL):
        self.interval = interval
        self._ensure_files()

    def _ensure_files(self):
        if not CP_FILE.exists():
            CP_FILE.write_text(json.dumps({
                "current_step": 0,
                "checkpoint_interval": self.interval,
                "last_checkpoint": 0,
                "status": "running",
                "pending_verification": False,
                "checkpoints": []
            }, indent=2))
        if not PRG_FILE.exists():
            PRG_FILE.write_text(json.dumps({"steps": [], "total_count": 0}, indent=2))

    def record_step(self, task_id: str, action: str, detail: str):
        """태스크 완료 시마다 step 기록"""
        cp  = json.loads(CP_FILE.read_text())
        prg = json.loads(PRG_FILE.read_text())

        cp["current_step"] += 1
        prg["total_count"] += 1
        prg["steps"].append({
            "step":    cp["current_step"],
            "task_id": task_id,
            "action":  action,   # "task_done" | "subtask_done" | "retry_success"
            "detail":  detail,
            "time":    datetime.now().isoformat()
        })

        CP_FILE.write_text(json.dumps(cp, indent=2))
        PRG_FILE.write_text(json.dumps(prg, indent=2))
        return cp["current_step"]

    def needs_checkpoint(self) -> bool:
        cp = json.loads(CP_FILE.read_text())
        steps_since = cp["current_step"] - cp["last_checkpoint"]
        return steps_since >= self.interval

    def request_verification(self, task_id: str, summary: list[str]):
        """검증 요청 등록 + 실행 일시 정지"""
        cp  = json.loads(CP_FILE.read_text())
        prg = json.loads(PRG_FILE.read_text())

        # 최근 N step 요약
        recent = prg["steps"][-self.interval:]

        cp_entry = {
            "checkpoint_no":  len(cp["checkpoints"]) + 1,
            "at_step":        cp["current_step"],
            "task_id":        task_id,
            "summary":        summary,
            "recent_steps":   recent,
            "requested_at":   datetime.now().isoformat(),
            "approved_at":    None,
            "approved_by":    "pending"
        }

        cp["checkpoints"].append(cp_entry)
        cp["last_checkpoint"]       = cp["current_step"]
        cp["pending_verification"]  = True
        cp["status"]                = "waiting_verification"

        CP_FILE.write_text(json.dumps(cp, indent=2))
        return cp_entry

    async def wait_for_approval(self, timeout: int = 600) -> bool:
        """
        승인 대기 (폴링)
        - 자동 승인 모드: AUTO_APPROVE=true 환경변수
        - 수동 승인: checkpoint.json status를 'approved'로 변경
        - timeout(초) 초과 시 자동 승인
        """
        import os
        if os.getenv("AUTO_APPROVE", "false").lower() == "true":
            self._approve("auto")
            return True

        print(f"\n{'═'*50}")
        print(f"  🔍 검증 요청 - shared/checkpoint.json 확인 후")
        print(f"  status 를 'approved' 또는 'rejected' 로 변경하세요")
        print(f"  타임아웃: {timeout}초")
        print(f"{'═'*50}\n")

        for _ in range(timeout):
            await asyncio.sleep(1)
            cp = json.loads(CP_FILE.read_text())
            if cp["status"] == "approved":
                print("  ✅ 검증 승인 - 계속 진행합니다")
                return True
            elif cp["status"] == "rejected":
                print("  ❌ 검증 거부 - 실행 중단")
                return False

        # 타임아웃 → 자동 승인
        print(f"  ⏱ 타임아웃 {timeout}s - 자동 승인")
        self._approve("timeout")
        return True

    def _approve(self, by: str = "manual"):
        cp = json.loads(CP_FILE.read_text())
        if cp["checkpoints"]:
            cp["checkpoints"][-1]["approved_at"] = datetime.now().isoformat()
            cp["checkpoints"][-1]["approved_by"] = by
        cp["status"]               = "running"
        cp["pending_verification"] = False
        CP_FILE.write_text(json.dumps(cp, indent=2))
